create or replace procedure public.raw_to_curated(source_stream varchar, target_table varchar, primary_key varchar)
returns varchar
language python
runtime_version = '3.11'
packages = ('snowflake-snowpark-python', 'requests')
handler = 'process_data'
as
$$

def process_data(session, source_stream, target_table, primary_key):
    table_def = session.sql(f"desc table {target_table}").collect()
    inner_merge_query = ""
    when_matched_query = ""
    when_not_matched_columns_query = ""
    when_not_matched_values_query = ""
    pk = primary_key.lower()
    
    for row in table_def:
        row_name = row["name"].lower()
        if row["type"] == 'TIMESTAMP_NTZ(9)':
            inner_merge_query = inner_merge_query + f'TO_TIMESTAMP_NTZ(AFTER:{row_name}::NUMBER/1000000) as {row_name},\n'
        else:
            inner_merge_query = inner_merge_query + f'AFTER:{row_name}::{row["type"]} as {row_name},\n'

        when_matched_query = when_matched_query + f'tgt.{row_name} = src.{row_name},\n'

        when_not_matched_columns_query = when_not_matched_columns_query + f'tgt.{row_name},'
        when_not_matched_values_query = when_not_matched_values_query + f'src.{row_name},'

    delete_sql = f"""
        DELETE FROM {target_table} tgt
        USING (
            SELECT
                BEFORE:{pk}::NUMBER AS {primary_key}
            FROM {source_stream}
            WHERE OP = 'd'
        ) src
        WHERE tgt.{primary_key} = src.{primary_key};
    """
    merge_sql = f"""
        MERGE INTO {target_table} tgt
        USING (
            SELECT * FROM (
                SELECT
                    {inner_merge_query[:-1]}
                    ROW_NUMBER() OVER(PARTITION BY AFTER:{pk}::NUMBER ORDER BY TS_MS DESC) as rn
                FROM {source_stream}
                WHERE OP IN ('c','r','u')
            ) WHERE rn = 1
        ) src
        ON src.{primary_key} = tgt.{primary_key}
        WHEN MATCHED THEN UPDATE SET
            {when_matched_query[:-2]}
        WHEN NOT MATCHED THEN INSERT ({when_not_matched_columns_query[:-1]})
            VALUES({when_not_matched_values_query[:-1]})
    """

    has_deletes = session.sql(f"""
        SELECT COUNT(*)
        FROM {source_stream}
        WHERE OP = 'd'
    """).collect()[0][0]

    if has_deletes:
        session.sql(delete_sql).collect()
    
    session.sql(merge_sql).collect()
    
    return f"Processed stream {source_stream}"

$$
;
